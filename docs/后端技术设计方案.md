```sql
CREATE TABLE `user` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '用户ID',

    `birthday` DATE DEFAULT NULL COMMENT '生日',
    `user_type` TINYINT NOT NULL DEFAULT 1 COMMENT '用户类型: 1-普通用户, 2-VIP用户, 3-企业用户',
    `user_level` INT DEFAULT 1 COMMENT '用户等级',
    `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 0-禁用, 1-正常, 2-待激活',
    `last_login_time` DATETIME DEFAULT NULL COMMENT '最后登录时间',
    `last_login_ip` VARCHAR(50) DEFAULT NULL COMMENT '最后登录IP',
    `login_fail_count` INT DEFAULT 0 COMMENT '连续登录失败次数',
    `lock_until` DATETIME DEFAULT NULL COMMENT '账户锁定截止时间',

    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    `deleted_at` DATETIME DEFAULT NULL COMMENT '删除时间',

    UNIQUE KEY `uk_username` (`username`),
    UNIQUE KEY `uk_phone` (`phone`),
    UNIQUE KEY `uk_email` (`email`),
    KEY `idx_status` (`status`),
    KEY `idx_user_type` (`user_type`),
    KEY `idx_created_at` (`created_at`),
    KEY `idx_inviter_id` (`inviter_id`)
```

#### 2.2.3.3 座位详情表 (seat)

座位表是座位图数据结构的核心，存储每个座位的详细信息：

```sql
CREATE TABLE `seat` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '座位ID',
    `session_id` BIGINT UNSIGNED NOT NULL COMMENT '场次ID',
    `venue_area_id` BIGINT UNSIGNED NOT NULL COMMENT '场馆区域ID',
    `venue_id` BIGINT UNSIGNED NOT NULL COMMENT '场馆ID',
    `area_name` VARCHAR(50) NOT NULL COMMENT '区域名称',
    `seat_no` VARCHAR(50) NOT NULL COMMENT '座位编号: 如A区-1排-10座',
    `seat_row` VARCHAR(10) NOT NULL COMMENT '排号',
    `seat_col` VARCHAR(10) NOT NULL COMMENT '座号',
    `price_level_id` BIGINT UNSIGNED NOT NULL COMMENT '票档ID',
    `price` DECIMAL(10,2) NOT NULL COMMENT '实际价格',
    `coordinate_x` DECIMAL(10,2) NOT NULL COMMENT '座位X坐标(像素)',
    `coordinate_y` DECIMAL(10,2) NOT NULL COMMENT '座位Y坐标(像素)',
    `coordinate_z` DECIMAL(10,2) DEFAULT 0 COMMENT '座位Z坐标(3D偏移)',
    `seat_width` INT DEFAULT 30 COMMENT '座位宽度(像素)',
    `seat_height` INT DEFAULT 30 COMMENT '座位高度(像素)',
    `rotation` DECIMAL(5,2) DEFAULT 0 COMMENT '座位旋转角度',
    `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 0-不可售, 1-可选, 2-锁定, 3-已预订, 4-已售出',
    `lock_expire_time` DATETIME DEFAULT NULL COMMENT '锁定过期时间',
    `lock_user_id` BIGINT UNSIGNED DEFAULT NULL COMMENT '锁定用户ID',
    `lock_session_id` VARCHAR(64) DEFAULT NULL COMMENT '锁定会话ID',
    `order_id` BIGINT UNSIGNED DEFAULT NULL COMMENT '订单ID',
    `lock_reason` VARCHAR(100) DEFAULT NULL COMMENT '锁定原因: user_lock/system_lock',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_session_seat` (`session_id`, `seat_no`),
    UNIQUE KEY `uk_session_coordinate` (`session_id`, `coordinate_x`, `coordinate_y`),
    KEY `idx_session_id` (`session_id`),
    KEY `idx_venue_area_id` (`venue_area_id`),
    KEY `idx_price_level_id` (`price_level_id`),
    KEY `idx_status` (`status`),
    KEY `idx_lock_expire` (`lock_expire_time`),
    KEY `idx_lock_user` (`lock_user_id`),
    KEY `idx_row_col` (`seat_row`, `seat_col`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='座位表';
```

**座位状态枚举**：

| 状态值 | 名称 | 说明 |
|--------|------|------|
| 0 | UNAVAILABLE | 不可售（通道、遮挡位等） |
| 1 | AVAILABLE | 可选（空闲可选） |
| 2 | LOCKED | 锁定（用户选座中） |
| 3 | RESERVED | 已预订（下单未支付） |
| 4 | SOLD | 已售出 |

### 3.3.4 座位图数据结构定义

座位图采用分层JSON结构，便于前端渲染和交互：

```typescript
// 座位图完整数据结构
interface SeatMapData {
  // 会话信息
  sessionInfo: {
    sessionId: number;
    sessionName: string;
    showDate: string;
    showTime: string;
  };
  
  // 座位图画布配置
  canvas: {
    width: number;      // 画布宽度
    height: number;     // 画布高度
    backgroundColor?: string;
    backgroundImage?: string;
  };
  
  // 缩放配置
  zoom: {
    min: number;         // 最小缩放
    max: number;         // 最大缩放
    default: number;    // 默认缩放
  };
  
  // 舞台信息
  stage: {
    x: number;
    y: number;
    width: number;
    height: number;
    label: string;
    color?: string;
  };
  
  // 区域列表
  areas: Area[];
  
  // 座位列表（扁平化，便于快速查询）
  seats: Seat[];
  
  // 统计信息
  statistics: {
    totalSeats: number;
    availableSeats: number;
    lockedSeats: number;
    soldSeats: number;
    unavailableSeats: number;
  };
}

// 区域数据
interface Area {
  areaId: number;
  areaCode: string;
  areaName: string;
  priceLevelId: number;
  price: number;
  
  // 区域渲染属性
  shape: 'rect' | 'arc' | 'polygon';
  
  // 矩形区域
  bounds?: {
    x: number;
    y: number;
    width: number;
    height: number;
  };
  
  // 弧形区域（扇形座位区）
  arcBounds?: {
    centerX: number;
    centerY: number;
    innerRadius: number;
    outerRadius: number;
    startAngle: number;  // 弧度
    endAngle: number;
  };
  
  // 多边形区域
  polygonPath?: Array<{x: number; y: number}>;
  
  // 样式
  styles: {
    fillColor: string;
    strokeColor: string;
    labelColor: string;
    labelFontSize: number;
  };
  
  // 标签位置
  labelPosition: {
    x: number;
    y: number;
  };
  
  // 区域座位数统计
  seatCount: {
    total: number;
    available: number;
    locked: number;
    sold: number;
  };
}

// 座位数据
interface Seat {
  seatId: number;
  seatNo: string;           // "A排-1座"
  seatRow: string;          // "A"
  seatCol: string;           // "1"
  
  // 坐标信息
  coordinate: {
    x: number;
    y: number;
    z?: number;              // 3D偏移
    rotation?: number;       // 旋转角度
  };
  
  // 尺寸
  size: {
    width: number;
    height: number;
  };
  
  // 归属区域
  areaId: number;
  areaName: string;
  
  // 价格
  priceLevelId: number;
  priceLevelName: string;
  price: number;
  
  // 状态
  status: SeatStatus;
  
  // 锁定信息（如果有）
  lockInfo?: {
    expireTime: string;
    remainSeconds: number;
    isOwnLock: boolean;      // 是否是当前用户锁定
  };
}

// 座位状态
enum SeatStatus {
  UNAVAILABLE = 0,   // 不可售
  AVAILABLE = 1,     // 可选
  LOCKED = 2,        // 锁定中
  RESERVED = 3,      // 已预订
  SOLD = 4           // 已售出
}
```

**详细流程说明**：

1. **选择座位**
   - 用户选择座位，提交座位ID列表
   - 校验座位是否可售

2. **库存校验与锁定**
   - 使用Redis分布式锁锁定座位
   - 锁定时长：15分钟
   - 检查座位状态是否为"可选"
   - 批量锁定座位，更新状态为"锁定"
   - 记录锁定超时时间

3. **创建订单**
   - 计算订单金额（座位价格之和 - 优惠）
   - 生成唯一订单号（时间戳+随机数）
   - 创建订单主记录和明细记录
   - 设置订单过期时间（30分钟）
   - 发布订单创建事件

4. **发起支付**
   - 调用支付服务预下单
   - 生成支付流水号
   - 返回支付链接/二维码
   - 记录支付记录

5. **支付处理**
   - 用户完成支付
   - 接收第三方异步回调
   - 验证签名
   - 更新支付状态
   - **支付成功**：确认订单，释放冗余座位锁定，更新票务销量
   - **支付超时**：自动取消订单，释放座位
   - **支付失败**：记录失败原因，保持座位锁定

```java
@Service
public class OrderServiceImpl implements OrderService {
    
    @Autowired
    private RedisLock redisLock;
    
    @Autowired
    private SeatService seatService;
    
    @Autowired
    private OrderRepository orderRepository;
    
    @Autowired
    private PaymentService paymentService;
    
    @Autowired
    private ApplicationEventPublisher eventPublisher;
    
    @Transactional(rollbackFor = Exception.class)
    public OrderCreateVO createOrder(OrderCreateDTO dto) {
        // 1. 校验用户
        User user = userService.getCurrentUser();
        
        // 2. 校验票务状态
        Ticket ticket = ticketService.getById(dto.getTicketId());
        Assert.isTrue(ticket.isOnSale(), "票务不在销售中");
        
        // 3. 获取座位信息
        List<Seat> seats = seatService.getByIds(dto.getSeatIds());
        
        // 4. 分布式锁锁定座位
        String lockKey = "seat:lock:" + dto.getSessionId();
        try {
            boolean locked = redisLock.tryLock(lockKey, 
                StringUtils.join(dto.getSeatIds(), ","), 
                15, TimeUnit.MINUTES);
            Assert.isTrue(locked, "座位锁定失败，请重试");
            
            // 5. 校验座位状态
            for (Seat seat : seats) {
                Assert.isTrue(seat.isAvailable(), "座位 {} 不可售", seat.getSeatNo());
            }
            
            // 6. 批量锁定座位
            seatService.lockSeats(seats, user.getId());
            
            // 7. 计算价格
            BigDecimal totalAmount = seats.stream()
                .map(Seat::getPrice)
                .reduce(BigDecimal.ZERO, BigDecimal::add);
            
            // 8. 校验并使用优惠券
            BigDecimal discountAmount = BigDecimal.ZERO;
            if (dto.getCouponId() != null) {
                discountAmount = couponService.useCoupon(dto.getCouponId(), user.getId(), totalAmount);
            }
            
            BigDecimal payAmount = totalAmount.subtract(discountAmount);
            
            // 9. 创建订单
            Order order = Order.builder()
                .orderNo(generateOrderNo())
                .userId(user.getId())
                .ticketId(dto.getTicketId())
                .sessionId(dto.getSessionId())
                .totalAmount(totalAmount)
                .discountAmount(discountAmount)
                .payAmount(payAmount)
                .orderStatus(OrderStatus.PENDING_PAYMENT)
                .expireTime(LocalDateTime.now().plusMinutes(30))
                .build();
            
            orderRepository.save(order);
            
            // 10. 创建订单明细
            List<OrderItem> items = seats.stream().map(seat -> 
                OrderItem.builder()
                    .orderId(order.getId())
                    .seatId(seat.getId())
                    .price(seat.getPrice())
                    .build()
            ).collect(Collectors.toList());
            orderItemRepository.saveAll(items);
            
            // 11. 发布订单创建事件
            eventPublisher.publishEvent(new OrderCreatedEvent(order));
            
            // 12. 发起支付
            PaymentVO payment = paymentService.createPayment(order);
            
            return OrderCreateVO.builder()
                .orderId(order.getId())
                .orderNo(order.getOrderNo())
                .payAmount(payAmount)
                .expireTime(order.getExpireTime())
                .payUrl(payment.getPayUrl())
                .build();
                
        } finally {
            // 注意：这里不释放锁，支付成功后再释放
        }
    }
}
```

#### 4.1.1.2 Redis座位锁定实现

```java
@Service
public class SeatLockServiceImpl implements SeatLockService {
    
    @Autowired
    private RedissonClient redissonClient;
    
    @Autowired
    private SeatRepository seatRepository;
    
    @Autowired
    private SeatLockMapper seatLockMapper;
    
    // 座位锁定时间（分钟）
    private static final int LOCK_TIMEOUT_MINUTES = 15;
    
    // Lua脚本：原子性检查和锁定座位
    private static final String LOCK_SEAT_SCRIPT = 
        "redis.call('SELECT', ARGV[1]) " +
        "local seatIds = cjson.decode(ARGV[2]) " +
        "local userId = ARGV[3] " +
        "local expireTime = ARGV[4] " +
        "local lockToken = ARGV[5] " +
        "local success = {} " +
        "local failed = {} " +
        "for i, seatId in ipairs(seatIds) do " +
        "    local key = 'seat:lock:' .. seatId " +
        "    local current = redis.call('GET', key) " +
        "    if current then " +
        "        table.insert(failed, seatId) " +
        "    else " +
        "        redis.call('SET', key, userId .. ':' .. lockToken, 'EX', " + LOCK_TIMEOUT_MINUTES * 60 + ") " +
        "        table.insert(success, seatId) " +
        "    end " +
        "end " +
        "return cjson.encode({success = success, failed = failed})";
    
    /**
     * 锁定座位
     */
    @Transactional(rollbackFor = Exception.class)
    public SeatLockResult lockSeats(SeatLockRequest request, Long userId) {
        Long sessionId = request.getSessionId();
        List<Long> seatIds = request.getSeatIds();
        
        // 1. 校验限购数量
        if (seatIds.size() > 4) {
            throw new BusinessException("单次最多选择4个座位");
        }
        
        // 2. 生成锁定令牌
        String lockToken = UUID.randomUUID().toString().replace("-", "");
        LocalDateTime expireTime = LocalDateTime.now().plusMinutes(LOCK_TIMEOUT_MINUTES);
        
        // 3. 获取座位信息
        List<Seat> seats = seatRepository.findBySessionIdAndIdIn(sessionId, seatIds);
        if (seats.size() != seatIds.size()) {
            throw new BusinessException("部分座位不存在");
        }
        
        // 4. 校验座位状态
        for (Seat seat : seats) {
            if (seat.getStatus() != SeatStatus.AVAILABLE) {
                throw new BusinessException("座位 " + seat.getSeatNo() + " 不可选");
            }
        }
        
        // 5. 校验同一区域或相邻区域
        validateSeatProximity(seats);
        
        // 6. Redis分布式锁（防止并发竞争）
        String globalLockKey = "seat:global:lock:" + sessionId;
        RLock globalLock = redissonClient.getLock(globalLockKey);
        boolean locked = false;
        
        try {
            locked = globalLock.tryLock(5, 30, TimeUnit.SECONDS);
            if (!locked) {
                throw new BusinessException("系统繁忙，请稍后重试");
            }
            
            // 7. 使用Redis原子操作锁定座位
            String seatIdsJson = JSON.toJSONString(seatIds);
            String dbIndex = "0"; // Redis数据库索引
            
            // 执行Lua脚本原子操作
            RedisScript<String> script = RedisScript.of(LOCK_SEAT_SCRIPT);
            String result = redisTemplate.execute(script, 
                List.of(dbIndex), 
                dbIndex, seatIdsJson, userId.toString(), 
                expireTime.toString(), lockToken);
            
            JSONObject resultObj = JSONObject.parseObject(result);
            JSONArray successIds = resultObj.getJSONArray("success");
            JSONArray failedIds = resultObj.getJSONArray("failed");
            
            if (failedIds.size() > 0) {
                // 释放已锁定的座位
                for (int i = 0; i < successIds.size(); i++) {
                    Long seatId = successIds.getLong(i);
                    redisTemplate.delete("seat:lock:" + seatId);
                }
                throw new BusinessException("部分座位已被其他用户锁定");
            }
            
            // 8. 更新MySQL座位状态
            LocalDateTime now = LocalDateTime.now();
            for (Long seatId : seatIds) {
                Seat seat = seats.stream()
                    .filter(s -> s.getId().equals(seatId))
                    .findFirst().orElseThrow();
                    
                seat.setStatus(SeatStatus.LOCKED);
                seat.setLockUserId(userId);
                seat.setLockSessionId(lockToken);
                seat.setLockExpireTime(expireTime);
                seat.setLockReason("user_lock");
            }
            seatRepository.saveAll(seats);
            
            // 9. 记录锁定日志
            recordLockLog(seatIds, userId, lockToken, expireTime, "LOCK");
            
            // 10. 构建返回结果
            BigDecimal totalAmount = seats.stream()
                .map(Seat::getPrice)
                .reduce(BigDecimal.ZERO, BigDecimal::add);
            
            return SeatLockResult.builder()
                .lockToken(lockToken)
                .lockExpireTime(expireTime)
                .lockSeats(seats.stream().map(seat -> 
                    SeatLockInfo.builder()
                        .seatId(seat.getId())
                        .seatNo(seat.getSeatNo())
                        .price(seat.getPrice())
                        .remainSeconds(LOCK_TIMEOUT_MINUTES * 60)
                        .build()
                ).collect(Collectors.toList()))
                .totalAmount(totalAmount)
                .expireSeconds(LOCK_TIMEOUT_MINUTES * 60)
                .build();
                
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new BusinessException("锁定超时，请重试");
        } finally {
            if (locked && globalLock.isHeldByCurrentThread()) {
                globalLock.unlock();
            }
        }
    }
    
    /**
     * 释放座位
     */
    @Transactional(rollbackFor = Exception.class)
    public void unlockSeats(SeatUnlockRequest request, Long userId) {
        String lockToken = request.getLockToken();
        List<Long> seatIds = request.getSeatIds();
        
        // 1. 校验锁定令牌
        String globalLockKey = "seat:global:lock:" + request.getSessionId();
        RLock globalLock = redissonClient.getLock(globalLockKey);
        
        try {
            globalLock.lock();
            
            // 2. 批量释放Redis锁
            for (Long seatId : seatIds) {
                String lockKey = "seat:lock:" + seatId;
                String lockValue = redisTemplate.opsForValue().get(lockKey);
                
                if (lockValue != null && lockValue.startsWith(userId + ":")) {
                    redisTemplate.delete(lockKey);
                }
            }
            
            // 3. 更新MySQL座位状态
            List<Seat> seats = seatRepository.findBySessionIdAndIdIn(
                request.getSessionId(), seatIds);
            
            for (Seat seat : seats) {
                seat.setStatus(SeatStatus.AVAILABLE);
                seat.setLockUserId(null);
                seat.setLockSessionId(null);
                seat.setLockExpireTime(null);
                seat.setLockReason(null);
            }
            seatRepository.saveAll(seats);
            
            // 4. 记录释放日志
            recordLockLog(seatIds, userId, lockToken, null, "UNLOCK");
            
        } finally {
            globalLock.unlock();
        }
    }
    
    /**
     * 校验座位临近关系
     */
    private void validateSeatProximity(List<Seat> seats) {
        if (seats.size() <= 1) return;
        
        // 按区域分组
        Map<Long, List<Seat>> areaSeats = seats.stream()
            .collect(Collectors.groupingBy(Seat::getVenueAreaId));
        
        // 校验同一区域座位是否连续
        for (Map.Entry<Long, List<Seat>> entry : areaSeats.entrySet()) {
            List<Seat> areaSeatList = entry.getValue();
            if (areaSeatList.size() > 1) {
                // 校验是否为连续座位
                List<String> rows = areaSeatList.stream()
                    .map(Seat::getSeatRow)
                    .distinct()
                    .sorted()
                    .collect(Collectors.toList());
                
                List<String> cols = areaSeatList.stream()
                    .map(Seat::getSeatCol)
                    .sorted()
                    .collect(Collectors.toList());
                
                // 简化校验：同排座位必须连续
                if (rows.size() > 1 && cols.size() > 1) {
                    // 跨排座位需要单独校验是否有空位
                }
            }
        }
    }
    
    /**
     * 记录锁定日志
     */
    private void recordLockLog(List<Long> seatIds, Long userId, 
                               String lockToken, LocalDateTime expireTime, 
                               String operation) {
        for (Long seatId : seatIds) {
            SeatLockLog log = SeatLockLog.builder()
                .seatId(seatId)
                .userId(userId)
                .lockToken(lockToken)
                .operation(operation)
                .expireTime(expireTime)
                .clientIp(getClientIp())
                .build();
            seatLockMapper.insert(log);
        }
    }
}
```

```java
@Service
public class SeatLockScheduler {
    
    @Autowired
    private SeatRepository seatRepository;
    
    @Autowired
    private SeatLockService seatLockService;
    
    @Autowired
    private NotifyService notifyService;
    
    /**
     * 每分钟执行：清理超时锁定的座位
     */
    @Scheduled(fixedRate = 60000)
    @Transactional
    public void cleanupExpiredLocks() {
        LocalDateTime now = LocalDateTime.now();
        
        // 1. 查询超时座位
        List<Seat> expiredSeats = seatRepository
            .findByStatusAndLockExpireTimeLessThan(
                SeatStatus.LOCKED, now);
        
        if (expiredSeats.isEmpty()) {
            return;
        }
        
        // 2. 批量释放
        List<Long> seatIds = expiredSeats.stream()
            .map(Seat::getId)
            .collect(Collectors.toList());
        
        // 更新数据库状态
        seatRepository.updateStatusBatch(seatIds, 
            SeatStatus.LOCKED, SeatStatus.AVAILABLE);
        
        // 清理Redis锁
        for (Long seatId : seatIds) {
            redisTemplate.delete("seat:lock:" + seatId);
        }
        
        // 3. 记录日志
        log.info("清理超时座位锁定，数量: {}", seatIds.size());
        
        // 4. 发送通知（可选）
        Set<Long> userIds = expiredSeats.stream()
            .map(Seat::getLockUserId)
            .filter(Objects::nonNull)
            .collect(Collectors.toSet());
        
        for (Long userId : userIds) {
            notifyService.sendSeatLockExpiredNotify(userId);
        }
    }
    
    /**
     * 每5分钟执行：发送锁定即将到期提醒
     */
    @Scheduled(cron = "0 */5 * * * ?")
    public void sendLockExpireWarning() {
        LocalDateTime warningTime = LocalDateTime.now().plusMinutes(5);
        LocalDateTime expireTime = LocalDateTime.now().plusMinutes(6);
        
        // 查询5分钟后即将过期的锁定
        List<Seat> warningSeats = seatRepository
            .findByStatusAndLockExpireTimeBetween(
                SeatStatus.LOCKED, warningTime, expireTime);
        
        // 按用户分组
        Map<Long, List<Seat>> userSeats = warningSeats.stream()
            .filter(s -> s.getLockUserId() != null)
            .collect(Collectors.groupingBy(Seat::getLockUserId));
        
        for (Map.Entry<Long, List<Seat>> entry : userSeats.entrySet()) {
            Long userId = entry.getKey();
            List<Seat> seats = entry.getValue();
            
            // 发送WebSocket通知
            notifyService.sendSeatLockWarningNotify(
                userId, 
                seats.size(),
                300 // 剩余秒数
            );
        }
    }
    
    /**
     * 启动时同步Redis和MySQL状态
     */
    @PostConstruct
    public void syncLockStatus() {
        log.info("开始同步座位锁定状态...");
        
        // 清理孤儿锁（Redis有锁但用户已下线）
        Set<String> lockKeys = redisTemplate.keys("seat:lock:*");
        if (lockKeys != null) {
            for (String key : lockKeys) {
                String lockValue = redisTemplate.opsForValue().get(key);
                if (lockValue != null) {
                    Long userId = Long.parseLong(lockValue.split(":")[0]);
                    
                    // 检查用户是否在线（通过Session）
                    if (!isUserOnline(userId)) {
                        Long seatId = extractSeatId(key);
                        redisTemplate.delete(key);
                        seatRepository.updateStatus(seatId, 
                            SeatStatus.AVAILABLE);
                        log.info("清理孤儿锁，seatId: {}", seatId);
                    }
                }
            }
        }
        
        log.info("座位锁定状态同步完成");
    }
```

```java
@Service
public class RefundServiceImpl implements RefundService {

        // 1. 获取订单
        Order order = orderRepository.findByOrderNo(dto.getOrderNo());
        
        // 2. 校验退款条件
        Assert.notNull(order, "订单不存在");
        Assert.isTrue(order.isPaid(), "订单未支付");
        Assert.isTrue(order.canRefund(), "订单不可退款");

        LocalDateTime showTime = order.getShowDateTime();
        long hoursBeforeShow = ChronoUnit.HOURS.between(LocalDateTime.now(), showTime);
        Assert.isTrue(hoursBeforeShow > 24, "演出前24小时内不支持退款");
        
        // 4. 计算退款金额
        BigDecimal refundAmount = calculateRefundAmount(order, hoursBeforeShow);
        
        // 5. 创建退款记录
        Refund refund = Refund.builder()
            .refundNo(generateRefundNo())
            .orderId(order.getId())
            .refundAmount(refundAmount)
            .refundReasonType(dto.getRefundReasonType())
            .refundReason(dto.getRefundReason())
            .refundStatus(RefundStatus.PENDING)
            .build();
        refundRepository.save(refund);
        
        // 6. 更新订单状态
        order.setRefundStatus(RefundStatus.PENDING);
        orderRepository.save(order);
        
        // 7. 触发退款审核流程
        refundWorkflowService.startAudit(refund);
        
        return RefundResultVO.builder()
            .refundNo(refund.getRefundNo())
            .refundAmount(refundAmount)
            .refundStatus(refund.getRefundStatus())
            .build();
    }
    
    private BigDecimal calculateRefundAmount(Order order, long hoursBeforeShow) {
        BigDecimal payAmount = order.getPayAmount();
        
        if (hoursBeforeShow > 168) { // >7天
            return payAmount; // 全额退款
        } else if (hoursBeforeShow > 72) { // 3-7天
            return payAmount.multiply(new BigDecimal("0.9")); // 90%
        } else { // 1-3天
            return payAmount.multiply(new BigDecimal("0.8")); // 80%
        }
    }
    
    @Transactional(rollbackFor = Exception.class)
    public void executeRefund(Refund refund) {
        try {
            // 1. 调用支付渠道退款
            Payment payment = paymentRepository.findByOrderId(refund.getOrderId());
            paymentService.refund(payment, refund.getRefundAmount());
            
            // 2. 更新退款状态
            refund.setRefundStatus(RefundStatus.COMPLETED);
            refund.setRefundTime(LocalDateTime.now());
            refundRepository.save(refund);
            
            // 3. 更新订单状态
            Order order = orderRepository.findById(refund.getOrderId());
            order.setRefundStatus(RefundStatus.COMPLETED);
            orderRepository.save(order);
            
            // 4. 释放座位
            seatService.releaseSeats(order);
            
            // 5. 更新票务销量
            ticketService.decrementSaleCount(order.getTicketId(), order.getTotalQuantity());
            
            // 6. 发送通知
            notifyService.sendRefundSuccessNotify(order.getUserId(), refund);
            
        } catch (Exception e) {
            refund.setRefundStatus(RefundStatus.FAILED);
            refund.setFailReason(e.getMessage());
            refundRepository.save(refund);
            throw e;
        }
    }
}
```

```java
@Service
public class InventoryServiceImpl implements InventoryService {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    @Autowired
    private InventoryRepository inventoryRepository;
    
    private static final String STOCK_KEY_PREFIX = "stock:ticket:";
    private static final String STOCK_LOCK_PREFIX = "stock:lock:";
    
    /**
     * 预扣减库存（Redis）
     */
    public boolean preDeductStock(Long ticketId, Long priceLevelId, int quantity) {
        String stockKey = STOCK_KEY_PREFIX + ticketId + ":" + priceLevelId;
        String lockKey = STOCK_LOCK_PREFIX + ticketId + ":" + priceLevelId;
        
        // 1. 检查并扣减库存
        Long stock = redisTemplate.opsForValue().decrement(stockKey, quantity);
        if (stock == null || stock < 0) {
            // 库存不足，补偿
            redisTemplate.opsForValue().increment(stockKey, quantity);
            return false;
        }
        
        // 2. 设置库存锁定标记（防止超卖）
        String lockValue = UUID.randomUUID().toString();
        redisTemplate.opsForValue().set(lockKey, lockValue, 15, TimeUnit.MINUTES);
        
        return true;
    }
    
    /**
     * 确认扣减库存（MySQL）
     */
    @Transactional
    public void confirmDeductStock(Long ticketId, Long priceLevelId, int quantity, String orderNo) {
        // 1. 扣减数据库库存
        int affected = inventoryRepository.deductStock(ticketId, priceLevelId, quantity);
        Assert.isTrue(affected > 0, "库存扣减失败");
        
        // 2. 记录库存流水
        InventoryLog log = InventoryLog.builder()
            .ticketId(ticketId)
            .priceLevelId(priceLevelId)
            .changeType(ChangeType.DEDUCT)
            .beforeStock(0) // 查询获取
            .changeNum(quantity)
            .afterStock(0)
            .orderNo(orderNo)
            .build();
        inventoryLogRepository.save(log);
        
        // 3. 释放Redis锁定
        String lockKey = STOCK_LOCK_PREFIX + ticketId + ":" + priceLevelId;
        redisTemplate.delete(lockKey);
    }
    
    /**
     * 释放预扣库存
     */
    public void releaseStock(Long ticketId, Long priceLevelId, int quantity) {
        String stockKey = STOCK_KEY_PREFIX + ticketId + ":" + priceLevelId;
        redisTemplate.opsForValue().increment(stockKey, quantity);
        
        String lockKey = STOCK_LOCK_PREFIX + ticketId + ":" + priceLevelId;
        redisTemplate.delete(lockKey);
    }
    
    /**
     * 同步Redis库存到MySQL（启动时/定时）
     */
    @Scheduled(cron = "0 */5 * * * ?")
    public void syncStockToRedis() {
        List<TicketPriceLevel> levels = priceLevelRepository.findAll();
        for (TicketPriceLevel level : levels) {
            String stockKey = STOCK_KEY_PREFIX + level.getTicketId() + ":" + level.getId();
            redisTemplate.opsForValue().set(stockKey, level.getAvailableStock());
        }
    }
    
    /**
     * 定时任务：清理超时预扣库存
     */
    @Scheduled(fixedRate = 60000)
    public void cleanupExpiredStock() {
        // 扫描超时预扣记录，释放冗余库存
        List<ExpiredStock> expiredList = findExpiredPreDeductRecords();
        for (ExpiredStock expired : expiredList) {
            releaseStock(expired.getTicketId(), expired.getPriceLevelId(), expired.getQuantity());
        }
    }
}
```

```java
public enum OrderStatus {
    PENDING_PAYMENT(1, "待支付"),
    PAID(2, "已支付"),
    COMPLETED(3, "已完成"),
    CANCELLED(4, "已取消"),
    REFUNDED(5, "已退款"),
    REFUNDING(6, "退款中");
    
    // 状态转换校验
    public static boolean canTransition(OrderStatus from, OrderStatus to) {
        Map<OrderStatus, Set<OrderStatus>> transitions = new HashMap<>();
        transitions.put(PENDING_PAYMENT, Set.of(PAID, CANCELLED));
        transitions.put(PAID, Set.of(COMPLETED, CANCELLED, REFUNDING));
        transitions.put(COMPLETED, Set.of(REFUNDING));
        transitions.put(REFUNDING, Set.of(REFUNDED));
        transitions.put(CANCELLED, Set.of());
        transitions.put(REFUNDED, Set.of());
        
        Set<OrderStatus> allowed = transitions.get(from);
        return allowed != null && allowed.contains(to);
    }
}

@Service
public class OrderStateMachine {
    
    public void transition(Order order, OrderStatus newStatus) {
        OrderStatus currentStatus = order.getOrderStatus();
        
        // 校验状态转换
        Assert.isTrue(
            OrderStatus.canTransition(currentStatus, newStatus),
            "订单状态不可从 {} 转换为 {}",
            currentStatus.getName(),
            newStatus.getName()
        );
        
        // 执行转换
        order.setOrderStatus(newStatus);
        
        // 触发钩子
        switch (newStatus) {
            case PAID:
                onOrderPaid(order);
                break;
            case COMPLETED:
                onOrderCompleted(order);
                break;
            case CANCELLED:
                onOrderCancelled(order);
                break;
            case REFUNDING:
                onOrderRefunding(order);
                break;
            case REFUNDED:
                onOrderRefunded(order);
                break;
        }
    }
    
    private void onOrderPaid(Order order) {
        order.setPayTime(LocalDateTime.now());
        // 释放冗余座位锁定
        seatService.releaseUnusedLocks(order.getId());
        // 更新票务销量
        ticketService.incrementSaleCount(order.getTicketId(), order.getTotalQuantity());
        // 发送通知
        notifyService.sendOrderPaidNotify(order);
    }
    
    private void onOrderCancelled(Order order) {
        order.setCancelTime(LocalDateTime.now());
        // 释放座位
        seatService.releaseSeats(order);
        // 返还优惠券
        couponService.returnCoupon(order.getUserId(), order.getCouponId());
        // 发送通知
        notifyService.sendOrderCancelledNotify(order);
    }
    
    private void onOrderRefunded(Order order) {
        // 释放座位
        seatService.releaseSeats(order);
        // 更新票务销量
        ticketService.decrementSaleCount(order.getTicketId(), order.getTotalQuantity());
        // 发送通知
        notifyService.sendRefundSuccessNotify(order);
    }
```

```java
            if (!checkRateLimit(userLimitKey, "user", 10, 60)) {

            if (!checkOrderCreation(clientIp, userId)) {

                return;

                writeResponse(response, 400, "请先通过验证码");
                return;
            }
        }
        
        chain.doFilter(request, response);
    }
    
    private boolean checkRateLimit(String key, String type, 
                                   int maxCount, int windowSeconds) {
        String redisKey = REQUEST_COUNT_KEY + type + ":" + key;
        
        Long count = redisTemplate.opsForValue().increment(redisKey);
        
        if (count == 1) {
            redisTemplate.expire(redisKey, windowSeconds, TimeUnit.SECONDS);
        }
        
        return count <= maxCount;
    }

        String ipKey = "order:limit:ip:" + ip;
        String userKey = "order:limit:user:" + userId;

        Long userCount = redisTemplate.opsForValue().increment(userKey);

        if (userCount == 1) {
```

```java
    public RiskResult evaluate(HttpRequest request, User user) {
        RiskContext context = RiskContext.builder()
            .userId(user.getId())
            .ip(request.getClientIp())
            .deviceId(request.getHeader("X-Device-Id"))
            .userAgent(request.getHeader("User-Agent"))
            .requestUri(request.getRequestURI())

            .build();
        
        List<RiskSignal> signals = new ArrayList<>();
        
        for (RiskRule rule : riskRules) {
            RiskSignal signal = rule.evaluate(context);
            signals.add(signal);
            
            // 高风险直接拦截
            if (signal.getLevel() == RiskLevel.HIGH) {
                return RiskResult.builder()
                    .passed(false)
                    .level(RiskLevel.HIGH)
                    .reason(signal.getReason())
                    .action(RiskAction.BLOCK)
                    .build();
            }
        }
        
        // 计算综合风险分数
        int totalScore = signals.stream()
            .mapToInt(RiskSignal::getScore)
            .sum();
        
        RiskLevel level = determineLevel(totalScore);
        RiskAction action = determineAction(level);
        
        return RiskResult.builder()
            .passed(action != RiskAction.BLOCK)
            .level(level)
            .score(totalScore)
            .action(action)
            .signals(signals)
            .build();
    }
}
```

```java
    /**
     * 获取座位图数据（带缓存）
     */
    public SeatMapVO getSeatMap(Long sessionId, boolean forceRefresh) {
        String cacheKey = SEAT_MAP_KEY + sessionId;
        
        // 尝试从缓存获取
        if (!forceRefresh) {
            SeatMapVO cached = (SeatMapVO) redisTemplate.opsForValue().get(cacheKey);
            if (cached != null) {
                // 补充实时座位状态
                enrichRealTimeStatus(cached, sessionId);
                return cached;
            }
        }
        
        // 从数据库加载
        SeatMapVO seatMap = loadSeatMapFromDB(sessionId);
        
        // 存入缓存
        redisTemplate.opsForValue().set(cacheKey, seatMap, 
            SEAT_MAP_TTL, TimeUnit.SECONDS);
        
        return seatMap;
    }

     */

        VenueSeatMap venueSeatMap = venueSeatMapRepository
            .findByVenueIdAndStatus(session.getVenueId(), 1);

        List<VenueArea> areas = venueAreaRepository
            .findByVenueId(session.getVenueId());

    }
    
    /**

     */
    private void enrichRealTimeStatus(SeatMapVO seatMap, Long sessionId) {
        // 实时获取锁定和已售座位
        List<Seat> realtimeSeats = seatRepository
            .findBySessionIdAndStatusIn(sessionId, 
                Arrays.asList(SeatStatus.LOCKED, SeatStatus.SOLD));
        
        // 更新座位状态
        Map<Long, Seat> seatMap = realtimeSeats.stream()
            .collect(Collectors.toMap(Seat::getId, s -> s));
        
        for (SeatVO seat : seatMap.getSeats()) {
            Seat realtime = seatMap.get(seat.getSeatId());
            if (realtime != null) {
                seat.setStatus(realtime.getStatus());
                if (realtime.getStatus() == SeatStatus.LOCKED) {
                    seat.setLockInfo(SeatLockInfo.builder()
                        .lockUserId(realtime.getLockUserId())
                        .expireTime(realtime.getLockExpireTime())
                        .isOwnLock(realtime.getLockUserId() != null && 
                            realtime.getLockUserId().equals(getCurrentUserId()))
                        .build());
                }
            }
        }
        
        // 重新统计
        updateStatistics(seatMap);
    }
    
    /**
     * 更新座位状态缓存
     */
    public void updateSeatStatusCache(Long sessionId, List<Long> seatIds, 
                                     SeatStatus status) {
        String cacheKey = SEAT_STATUS_KEY + sessionId;
        
        for (Long seatId : seatIds) {
            String key = cacheKey + ":" + seatId;
            SeatStatusCache cache = new SeatStatusCache();
            cache.setSeatId(seatId);
            cache.setStatus(status);
            cache.setUpdateTime(System.currentTimeMillis());
            
            redisTemplate.opsForValue().set(key, cache, 
                SEAT_STATUS_TTL, TimeUnit.SECONDS);
        }
        
        // 清除座位图缓存（下次访问会重新加载）
        redisTemplate.delete(SEAT_MAP_KEY + sessionId);
    }
    
    /**
     * 批量预加载热门场次座位图
     */
    @Scheduled(fixedRate = 60000) // 每分钟
    public void preloadHotSessionSeatMaps() {
        // 获取即将开售的热门场次
        List<Long> hotSessionIds = getHotSessionIds();
        
        for (Long sessionId : hotSessionIds) {
            try {
                getSeatMap(sessionId, true);
                log.debug("预加载座位图, sessionId: {}", sessionId);
            } catch (Exception e) {
                log.error("预加载座位图失败, sessionId: {}", sessionId, e);
            }
        }
    }
```

```java
@Configuration
public class RateLimitConfig {
    
    @Bean
    public过滤器InterceptorRegistration<RateLimitInterceptor> rateLimitInterceptor() {
        InterceptorRegistration<RateLimitInterceptor> registration = 
            registry.addInterceptor(rateLimitInterceptor());
        
        registration.addPathPatterns("/api/**")
            .excludePathPatterns(
                "/api/v1/health",
                "/api/v1/captcha/image"
            );
        
        return registration;
    }
}

@Component
public class RateLimitInterceptor implements HandlerInterceptor {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    @Autowired
    private RateLimitRuleService rateLimitRuleService;
    
    @Override
    public boolean preHandle(HttpServletRequest request, 
                             HttpServletResponse response,
                             Object handler) throws Exception {
        
        String clientKey = getClientKey(request);
        String requestUri = request.getRequestURI();
        
        // 获取限流规则
        RateLimitRule rule = rateLimitRuleService.getRule(requestUri);
        
        if (rule == null) {
            return true;
        }
        
        String redisKey = "ratelimit:" + rule.getType() + ":" + clientKey + ":" + requestUri;
        
        // 使用Redis滑动窗口限流
        long currentTime = System.currentTimeMillis();
        long windowStart = currentTime - rule.getWindowMs();
        
        // 移除窗口外的记录
        redisTemplate.opsForZSet().removeRangeByScore(redisKey, 0, windowStart);
        
        // 统计当前窗口内的请求数
        Long count = redisTemplate.opsForZSet().zCard(redisKey);
        
        if (count != null && count >= rule.getMaxRequests()) {
            response.setStatus(429);
            response.setContentType("application/json;charset=UTF-8");
            response.getWriter().write(
                "{\"code\":429,\"message\":\"请求过于频繁，请稍后再试\"}");
            return false;
        }
        
        // 添加当前请求
        redisTemplate.opsForZSet().add(redisKey, 
            String.valueOf(currentTime), currentTime);
        redisTemplate.expire(redisKey, 
            rule.getWindowMs(), TimeUnit.MILLISECONDS);
        
        // 设置限流响应头
        response.setHeader("X-RateLimit-Limit", String.valueOf(rule.getMaxRequests()));
        response.setHeader("X-RateLimit-Remaining", 
            String.valueOf(rule.getMaxRequests() - count - 1));
        response.setHeader("X-RateLimit-Reset", 
            String.valueOf((currentTime + rule.getWindowMs()) / 1000));
        
        return true;
    }
}

@Service
public class RateLimitRuleService {
    
    public RateLimitRule getRule(String requestUri) {
        // 精确匹配
        RateLimitRule rule = redisTemplate.opsForHash()
            .get("ratelimit:rules:exact", requestUri, RateLimitRule.class);
        if (rule != null) {
            return rule;
        }
        
        // 前缀匹配
        Map<Object, Object> prefixRules = redisTemplate.opsForHash()
            .entries("ratelimit:rules:prefix");
        for (Map.Entry<Object, Object> entry : prefixRules.entrySet()) {
            String pattern = (String) entry.getKey();
            if (requestUri.startsWith(pattern)) {
                return (RateLimitRule) entry.getValue();
            }
        }
        
        // 默认规则
        return RateLimitRule.builder()
            .type("default")
            .maxRequests(100)
            .windowMs(60000)
            .build();
    }
}
```
